<!-- Contract that plan 04-04's merge orchestrator consumes. Define it first. -->

CanonicalVehicle {
  brand: string;            // display-cased, ADEME spelling
  model: string;
  fuelType: FuelType;
  consumption: number;      // L/100km (ICE) or kWh/100km (EV) — already converted
  batteryCapacityKwh: number | null;
  tankCapacityLiters: number | null;
  source: string;           // 'ademe' | 'epa'
}

CatalogSourceAdapter {
  readonly source: string;          // provenance tag written to the row
  readonly precedence: number;      // 0 = highest (ADEME); 10 = EPA; a new source picks a number
  load(): Promise<unknown[]>;       // ADEME: live paginated fetch; EPA: read committed snapshot
  normalize(raw: unknown): CanonicalVehicle | null;  // convert + map + filter; null = skip
}
</interfaces>
